            background: 'linear-gradient(145deg, #1e293b 0%, #0f172a 100%)',
            pt: 2,
            zIndex: (theme) => theme.zIndex.appBar + 10,
          },
        }}
      >
        <Box
          sx={{
            display: 'flex',
            justifyContent: 'space-between',
            px: 3,
            mb: 3,
          }}
        >
          <YTXTypography
            variant="h6"
            sx={{
              background: 'linear-gradient(135deg, #00F5FF 0%, #0099FF 100%)',
              WebkitBackgroundClip: 'text',
              WebkitTextFillColor: 'transparent',
              fontWeight: 800,
            }}
          >
            Yantrix
          </YTXTypography>
          <IconButton onClick={() => setMobileOpen(false)} sx={{ color: '#00F5FF' }}>
            <Close />
          </IconButton>
        </Box>

        <List>
          {submenuStructure.map((item) => (
            <Box key={item.label}>
              {item.items ? (
                <>
                  <ListItem
                    button
                    onClick={() => toggleMobileSubmenu(item.label)}
                    sx={{
                      color: activeTop === item.label ? '#00F5FF' : 'rgba(255, 255, 255, 0.9)',
                      py: 2,
                      px: 3,
                      background:
                        activeTop === item.label ? 'rgba(0, 245, 255, 0.08)' : 'transparent',
                    }}
                  >
                    <YTXTypography sx={{ fontWeight: 600, fontSize: '16px' }}>
                      {item.label}
                    </YTXTypography>
                    <KeyboardArrowDown
                      sx={{
                        transform: mobileSubmenu[item.label] ? 'rotate(180deg)' : 'rotate(0)',
                        transition: 'transform 0.3s',
                      }}
                    />
                  </ListItem>
                  <Collapse in={mobileSubmenu[item.label]} timeout="auto">
                    <List sx={{ pl: 2, background: 'rgba(0, 0, 0, 0.2)' }}>
                      {item.items.map((subItem, index) => (
                        <ListItem
                          button
                          key={index}
                          onClick={() => handleSubItemClick(item.label, subItem.href)}
                          sx={{
                            color: 'rgba(255, 255, 255, 0.75)',
                            py: 1.5,
                            pl: 4,
                            '&:hover': { color: '#00F5FF' },
                          }}
                        >
                          <YTXTypography sx={{ fontSize: '14px' }}>{subItem.label}</YTXTypography>
                        </ListItem>
                      ))}
                    </List>
                  </Collapse>
                </>
              ) : (
                <ListItem
                  button
                  onClick={() => handleTopItemClick(item.label, item.href)}
                  sx={{
                    color: activeTop === item.label ? '#00F5FF' : 'rgba(255, 255, 255, 0.9)',
                    py: 2,
                    px: 3,
                    background:
                      activeTop === item.label ? 'rgba(0, 245, 255, 0.08)' : 'transparent',
                  }}
                >
                  <YTXTypography sx={{ fontWeight: 600, fontSize: '16px' }}>
                    {item.label}
                  </YTXTypography>
                </ListItem>
              )}
            </Box>
          ))}
        </List>

        <Box sx={{ px: 3, mt: 3 }}>
          <Button
            fullWidth
            variant="contained"
            sx={{
              background: 'linear-gradient(135deg, #00F5FF 0%, #0099FF 100%)',
              color: '#0f172a',
              fontWeight: 700,
              py: 1.5,
              borderRadius: '12px',
              textTransform: 'none',
              fontSize: '16px',
            }}
          >
            Get Started
          </Button>
        </Box>
      </Drawer>
    </>
  );
}
